<good_example>
DO THIS - simple input, many API calls:

```c
int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 4) return 0;

  // Simple: pass fuzzer data directly to parser
  void* ctx = lib_parse(data, size);     // Parser API
  if (!ctx) return 0;

  lib_validate(ctx);                      // +1 API
  lib_get_count(ctx);                     // +1 API
  lib_get_item(ctx, 0);                   // +1 API
  lib_iterate(ctx);                       // +1 API
  lib_serialize(ctx, NULL, 0);            // +1 API

  lib_free(ctx);
  return 0;
}
// Result: 15 lines total, 6 library API calls = HIGH COVERAGE
```
</good_example>
