<good_example>
DO THIS - simple input, many API calls:

```cpp
#include <stdint.h>
#include <fuzzer/FuzzedDataProvider.h>
#include "library_header.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider fdp(data, size);

  // Simple input - just get bytes from fuzzer
  std::string input = fdp.ConsumeRandomLengthString(4096);

  // Call MANY library APIs
  void* ctx = lib_create();
  if (!ctx) return 0;

  lib_parse(ctx, input.data(), input.size());  // Parser API
  lib_validate(ctx);                            // +1 API
  lib_get_count(ctx);                           // +1 API
  lib_get_item(ctx, 0);                         // +1 API
  lib_iterate(ctx);                             // +1 API
  lib_serialize(ctx);                           // +1 API

  lib_destroy(ctx);
  return 0;
}
// Result: 20 lines total, 7 library API calls = HIGH COVERAGE
```
</good_example>
