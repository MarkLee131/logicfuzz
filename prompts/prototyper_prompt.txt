    switch (fdp.ConsumeIntegralInRange<int>(0, 4)) {
      case 0: return "null";
      case 1: return fdp.ConsumeBool() ? "true" : "false";
      case 2: return std::to_string(fdp.ConsumeIntegral<int>());

    std::string k = fdp.ConsumeRandomLengthString(8);
    if (k.empty()) k = "key";
    return "{\"" + k + "\":" + MakeJson(fdp, depth-1) + "}";
  } else {  // array
    return "[" + MakeJson(fdp, depth-1) + "]";
  }
}
std::string json = MakeJson(fdp, fdp.ConsumeIntegralInRange<int>(1, 4));
